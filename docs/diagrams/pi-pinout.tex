    pinUsed/.style={pin, fill=cyan!30},

    pinFree/.style={pin, fill=white},
    pinConflict/.style={pin, fill=red!50, draw=red!75!black, line width=0.6mm},
    fnL/.style={anchor=east, font=\sffamily\small},

    callout/.style={anchor=west, font=\sffamily\footnotesize, inner sep=2pt},
  ]

  \def\colgap{18mm}

  % Pin definitions: row, leftStyle, leftFn, leftPinNum, rightStyle, rightFn, rightPinNum
  \foreach \r/\lstyle/\lfn/\lpin/\rstyle/\rfn/\rpin in {%

     2/pinFree/GPIO 2 (SDA)/3 /pinPower/5V/4,
     3/pinFree/GPIO 3 (SCL)/5 /pinGnd/GND/6,
     4/pinFree/GPIO 4/7   /pinFree/GPIO 14 (TX)/8,
     5/pinGnd/GND/9       /pinFree/GPIO 15 (RX)/10,
     6/pinFree/GPIO 17/11 /pinPlanned/GPIO 18/12,
     7/pinFree/GPIO 27/13 /pinGnd/GND/14,
     8/pinFree/GPIO 22/15 /pinFree/GPIO 23/16,
     9/pinPower/3.3V/17   /pinFree/GPIO 24/18,
    10/pinFree/GPIO 10/19 /pinGnd/GND/20,
    11/pinFree/GPIO 9/21  /pinFree/GPIO 25/22,
    12/pinFree/GPIO 11/23 /pinFree/GPIO 8/24,
    13/pinGnd/GND/25      /pinFree/GPIO 7/26,
    14/pinFree/GPIO 0/27  /pinFree/GPIO 1/28,
    15/pinFree/GPIO 5/29  /pinGnd/GND/30,
    16/pinUsed/GPIO 6/31  /pinFree/GPIO 12/32,
    17/pinUsed/GPIO 13/33 /pinGnd/GND/34,
    18/pinConflict/GPIO 19/35 /pinFree/GPIO 16/36,
    19/pinUsed/GPIO 26/37 /pinPlanned/GPIO 20/38,
    20/pinGnd/GND/39      /pinPlanned/GPIO 21/40%
  }{%
     \node[\lstyle] (Lp\r) at (0, -\r*\rowstep) {\lpin};
     \node[fnL] at ([xshift=-6mm]Lp\r.west) {\lfn};
     \node[\rstyle] (Rp\r) at (\colgap, -\r*\rowstep) {\rpin};
     \node[fnR] at ([xshift=6mm]Rp\r.east) {\rfn};
  }

  % header outline — frame tight to pins, labels stay outside it
  \node[draw=green!50!black, semithick, rounded corners=1pt, fit=(Lp1)(Rp20),
        inner sep=2mm, fill=green!3, on background layer] (header) {};

  \node[font=\sffamily\bfseries\large, above=3mm of header.north]
        {Raspberry Pi 40-pin GPIO header (top-down view)};

  %================================================================
  % CALLOUTS — wire colour for each used / planned pin
  % Place them with breathing room
  %================================================================
  \def\leftPad{38mm}
  \def\rightPad{38mm}

  % --- LEFT column callouts ---
  % pin 1 — 3V3 → mic VDD
  \node[callout, anchor=east] at ([xshift=-\leftPad]Lp1.west)
        (cVDD) {to \textbf{INMP441 VDD} \quad
                \textcolor{wireRed}{\rule{6mm}{3mm}} \textbf{red}};
  \draw[-{Stealth[length=2mm]}, thick, gray] (cVDD.east) -- (Lp1.west);

  % pin 31 GPIO6 IN4
  \node[callout, anchor=east] at ([xshift=-\leftPad]Lp16.west)
        (cIN4) {\textbf{IN4 motor} \quad
                \textcolor{wireGray}{\rule{6mm}{3mm}} \textbf{gray}};
  \draw[-{Stealth[length=2mm]}, thick, gray] (cIN4.east) -- (Lp16.west);

  % pin 33 GPIO13 IN3
  \node[callout, anchor=east] at ([xshift=-\leftPad]Lp17.west)
        (cIN3) {\textbf{IN3 motor} \quad
                \fcolorbox{black!40}{wireWhite}{\rule{6mm}{3mm}} \textbf{white}};
  \draw[-{Stealth[length=2mm]}, thick, gray] (cIN3.east) -- (Lp17.west);

  % pin 35 GPIO19 — IN2 now / LRCLK planned (CONFLICT)
  \node[callout, anchor=east, draw=red!60, dashed, rounded corners,

                \textcolor{wireGreen}{\rule{6mm}{3mm}} \textbf{green} (after rewire)};

        (cIN2.east) -- (Lp18.west);

  % pin 37 GPIO26 IN1
  \node[callout, anchor=east] at ([xshift=-\leftPad]Lp19.west)

  \draw[-{Stealth[length=2mm]}, thick, gray] (cIN1.east) -- (Lp19.west);

  % pin 2 — 5V → amp VIN
  \node[callout, anchor=west] at ([xshift=\rightPad]Rp1.east)

                \textcolor{wireOrange}{\rule{6mm}{3mm}} \textbf{orange}};

  \node[callout, anchor=west, fill=black!8, inner sep=3pt]

        (cGND) {\textbf{GND} (any pin) — common rail

                \textcolor{wireBlack}{\rule{6mm}{3mm}} \textbf{black}};

  \node[callout, anchor=west] at ([xshift=\rightPad]Rp6.east)

  \draw[-{Stealth[length=2mm]}, thick, gray] (Rp6.east) -- (cBCLK.west);

  % pin 38 GPIO 20 — I2S SD (mic data → Pi)
  \node[callout, anchor=west] at ([xshift=\rightPad]Rp19.east)

  \node[callout, anchor=west] at ([xshift=\rightPad]Rp20.east)
